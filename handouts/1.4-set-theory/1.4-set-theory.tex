\prob Prove that if $A\subset B$, then $B^c \subset A^c$.

\vfill
\prob Suppose that a number $x$ is to be selected from the real line $S$, and let $A$ and $B$ be the following events:

\begin{align*}
A &= \{x : 1\leq x \leq 3\}, \\
B &= \{ x : 3 \leq x \leq 7 \}.
\end{align*}

Describe each of the following events.

\begin{enumerate}
\item $A^c$\vfill
\item $A \cup B$\vfill
\item $A^c \cap B^c$\vfill
\end{enumerate}

\prob Suppose that a coin is tossed three times. Identify an appropriate sample space for this experiment, and then identify each of the events as subsets of the sample space.

\begin{enumerate}
\item The sample space $S$ is...\vfill

\item The event $B$ that a head is obtained on the second toss.\vfill
\item The event $C$ that a tail is obtained on the third toss.\vfill
\item The event $D$ that no heads are obtained.\vfill
\end{enumerate}
