\vfill
\prob Suppose that a discrete random variable $X$ has probability function
	\[
	f(x) = \begin{cases}
	\frac{c}{2^x} & : x=0,1,2,\ldots, \\
	0 & : \text{otherwise},
	\end{cases}
	\]
for some positive constant $c$.

\medskip
\begin{enumerate}
\item Find the value of $c$.\vfill
\item Find the probability of the event $C = \{0,3,83\}$.\vfill
\item Find the probability of the event $D = \{0,2,4,6,\ldots\}$.
\end{enumerate}
