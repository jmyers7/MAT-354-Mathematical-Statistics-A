\prob Suppose that $X$ is a continuous random variable with probability density function
	\[
	f(x) = \begin{cases}
	cx^2 & : 0 \leq x \leq 2, \\
	0 & : \text{otherwise},
	\end{cases}
	\]
where $c$ is a constant.

\begin{enumerate}
\item Compute $c$.\vfill
\item Compute $P( X \geq 1)$.\vfill
\item Compute $P( X \geq 1 \mid X \geq 0.5)$.
\end{enumerate}

\vfill
\prob Suppose that the probability density function of a random variable $X$ is given by
	\[
	f(x) = \begin{cases}
	\frac{1}{8}x & : 0 \leq x \leq 4, \\
	0 & : \text{otherwise}.
	\end{cases}
	\]
After a random value of $X$ is observed, let $Y$ be the integer closest to $X$. Find the probability function of $Y$.
