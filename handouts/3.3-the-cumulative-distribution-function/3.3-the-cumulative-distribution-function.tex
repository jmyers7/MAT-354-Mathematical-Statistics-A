\prob Let $X$ be a continuous random variable with density function
	\[
	f(x) = \begin{cases}
	2x & : 0 < x< 1, \\
	0 & : \text{else}.
	\end{cases}
	\]
Find its cumulative distribution function.

\vfill
\prob The length of time to failure (in hundreds of hours) for a transistor is a random variable $X$ with cumulative distribution function given by
	\[
	F(x) = \begin{cases}
	1-e^{-x^2} & : x\geq 0, \\
	0 & : \text{else}.
	\end{cases}
	\]
Compute the probability $P(X > 1 \mid X \leq 2)$.
