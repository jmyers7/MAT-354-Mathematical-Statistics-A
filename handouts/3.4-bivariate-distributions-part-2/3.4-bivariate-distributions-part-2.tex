\prob (Example 3.4.14) Suppose that $X$ and $Y$ are jointly continuous random variables with supports in the interval $[0,2]$. Suppose also that the joint cumulative distribution function has
	\[
	F(x,y) = \frac{1}{16}xy(x+y)
	\]
for $0 \leq x \leq 2$ and $0 \leq y \leq 2$.

\medskip
\begin{enumerate}
\item Compute a formula for the cumulative distribution function for every point $(x,y)$ in the plane.\vfill
\item Compute the cumulative distribution functions of $X$ and $Y$.\vfill
\item Compute the joint probability density function of $X$ and $Y$.\vfill
\end{enumerate}
