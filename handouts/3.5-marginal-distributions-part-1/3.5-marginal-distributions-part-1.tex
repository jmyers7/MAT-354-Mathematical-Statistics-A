\prob Suppose $X$ and $Y$ are jointly discrete random variables with joint probability function $f(x,y)$ given by
	\[
	\begin{array}{c|llll}
       	f(x,y) & y=1 & y=2 & y=3 & y=4 \\
	\hline
	x=1 & 0.1 & 0 & 0.1 & 0 \\
	x=2 & 0.3 & 0 & 0.1 & 0.2\\
	x=3 & 0 & 0.2 & 0 & 0\\
	\end{array}
	\]
Compute the marginal probability functions of $X$ and $Y$.

\vfill
\prob Suppose $X$ and $Y$ are jointly continuous random variables with joint probability density function $f(x,y)$ given by
	\[
	f(x,y) = \begin{cases}
	\frac{21}{4} x^2 y & : x^2 \leq y \leq 1, \\
	0 & : \text{else}.
	\end{cases}
	\]
Compute the marginal probability density functions of $X$ and $Y$.
