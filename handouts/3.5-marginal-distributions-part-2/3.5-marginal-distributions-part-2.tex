\prob Suppose $X$ and $Y$ are jointly continuous with probability density function
	\[
	f(x,y) = \begin{cases}
	6xy^2 & : 0 \leq x \leq 1 ,\ 0 \leq y \leq 1,\\
	0 & : \text{else}.
	\end{cases}
	\]

\begin{enumerate}
\item Check whether $X$ and $Y$ are independent using marginal probability density functions.\vfill
\item Check whether $X$ and $Y$ are independent using cumulative distribution functions.
\end{enumerate}

\vfill
\prob Suppose $X$ and $Y$ are jointly continuous with probability density function
	\[
	f(x,y) = \begin{cases}
	2 & : 0 \leq y \leq x \leq 1,\\
	0 & : \text{else}.
	\end{cases}
	\]
Are $X$ and $Y$ independent?

\vfill
\newpage
\prob Suppose $X$ and $Y$ are jointly discrete random variables with joint probability function $f(x,y)$ given by
	\[
	\begin{array}{c|lll}
       	f(x,y) & y=0 & y=1 & y=2 \\
	\hline
	x=0 & 0.38 & 0.14 & 0.24 \\
	x=1 & 0.17 & 0.02 & 0.05
	\end{array}
	\]
Are $X$ and $Y$ independent?
