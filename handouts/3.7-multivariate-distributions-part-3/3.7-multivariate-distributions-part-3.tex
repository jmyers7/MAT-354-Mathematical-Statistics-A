\prob Consider the follow p.d.f.\ on the real line:
\[
	f(x) = \begin{cases}
	c x^5 e^{-x} & : x > 0, \\
	0 & : \text{else},
	\end{cases}
\]
for some constant $c$.

\begin{enumerate}
\item Compute the constant $c$.\vfill
\item Let $X_1,\ldots,X_n$ be the heights (in feet) of a random sample of people chosen from our school. Explain why it is reasonable to believe that these random variables are independent.\vfill
\item Explain why $f(x)$ might be a reasonable p.d.f.\ for the $X_i$'s.\vfill
\item Suppose $n\geq 2$. What is the probability that the heights of the first and second people are both less than six feet?
\end{enumerate}

\prob Suppose a fair six-sided die is rolled $n$ times. Let $X_i$ be the result of the $i$-th roll. 

\begin{enumerate}
\item Explain why $X_1,\ldots,X_n$ is an i.i.d.\ sequence of random variables. What is their common distribution?\vfill
\item Compute the probability that each roll results in a $3$.\vfill
\end{enumerate}
